# ============================================================================
#  启动层 CMakeLists.txt —— startup 静态库
# ----------------------------------------------------------------------------
#  把"让 C 程序能跑起来"的最底层代码收进一个库:
#    startup_S32K144.S :中断向量表 + Reset_Handler,上电后第一段执行的代码
#    startup.c         :C 层启动逻辑,搬 .data、清 .bss,最后调 main
#    system_S32K144.c  :SystemInit(),配置时钟系统
#  收成库而不是直接把源文件塞给可执行 target,好处是:
#  启动代码的编译配置和头文件依赖被封装在这一层,应用层只管链接它。
# ============================================================================

# ----------------------------------------------------------------------------
# 定义静态库 target
#   STATIC:打进 .a 归档,链接时按需取用(裸机固件的标准做法)。
#   startup_S32K144.S 是大写 .S(CMake 按 ASM 语言处理):
#   大写 .S = 带预处理(gcc 会先展开 #include/#define 再汇编);
#   小写 .s = 直接汇编。S32DS 生成的启动文件用了很多预处理指令,
#   必须保持大写。
# ----------------------------------------------------------------------------
add_library(startup STATIC
    startup.c
    startup_S32K144.S
    system_S32K144.c
)

# ----------------------------------------------------------------------------
# 头文件路径的对外暴露 —— 这是"分层 + 依赖传播"的核心示范
#   INTERFACE 的含义:这个路径不用于编译 startup 自己,而是
#   "凡是链接我的人,自动获得这个 include 路径"。
#
#   所以应用层 main.c 里一句
#       target_link_libraries(s32k144.elf PRIVATE startup)
#   之后,#include "device_registers.h" 就直接能找到——
#   应用层的 CMakeLists 里不需要再出现任何 include_directories(),
#   头文件依赖跟着依赖关系自动流动。
#
#   ${PROJECT_SOURCE_DIR} = 顶层 CMakeLists.txt 所在目录
#   (即 ~/s32k144),用它是为了不受从哪个目录进入构建的影响。
# ----------------------------------------------------------------------------
target_include_directories(startup
    INTERFACE ${PROJECT_SOURCE_DIR}/include
)
